## 1.5 공통 스타일 규칙

**테이블 스타일:**
- 모든 테이블 헤더(TableHead > TableRow > TableCell): `align="center"`
- 테이블 레이아웃: `tableLayout: 'fixed'` (고정 컬럼 너비 적용 시)
- 컬럼 너비: 퍼센트(%) 기반 비율 설정

**상태 표시 (한글):**
```javascript
const getStatusName = (status) => {
  switch (status) {
    case 'AVAILABLE': return '사용가능';
    case 'MAINTENANCE': return '점검중';
    case 'UNAVAILABLE': return '사용불가';
    default: return status;
  }
};
```

**레이아웃:**
- 메인 콘텐츠 영역: 80% 너비, 중앙 정렬 (`maxWidth: '80%'`, `mx: 'auto'`)

    const day = startOfWeek.add(i, 'day');
    days.push({
      date: day.format('YYYY-MM-DD'),
      dayName: dayNames[i],
      dayNum: day.format('D'),
      isToday: day.isSame(dayjs(), 'day'),
      isSunday: i === 0,
      isSaturday: i === 6,
    });
  }
  return days;
};
```

    days.push({ date: null, dayNum: null });
  }
  // 현재 달 날짜
  for (let d = 1; d <= endOfMonth.date(); d++) {
    days.push({
      date: startOfMonth.date(d).format('YYYY-MM-DD'),
      dayNum: d,
    });
  }
  // 마지막 줄을 7개로 채우기
  while (days.length % 7 !== 0) {
    days.push(null);
  }
  return days;
};

const initialState = {
  user: JSON.parse(localStorage.getItem('user')) || null,
  accessToken: localStorage.getItem('accessToken') || null,

  isAuthenticated: !!localStorage.getItem('accessToken'),

```

**Actions:**
| Action | 설명 | Payload |
|--------|------|---------|
| login | 로그인 처리 | { accessToken, refreshToken, user } |
| logout | 로그아웃 처리 | 없음 |
| updateTokens | 토큰 갱신 | { accessToken, refreshToken } |
| updateUser | 사용자 정보 갱신 | { role, department, position, phone } |

**Reducers:**
```javascript
export const authSlice = createSlice({
  name: 'auth',
  initialState,
  reducers: {
    login: (state, action) => {
      const { accessToken, refreshToken, user } = action.payload;
      state.accessToken = accessToken;
      state.refreshToken = refreshToken;
      state.user = user;
      state.isAuthenticated = true;
      localStorage.setItem('accessToken', accessToken);
      localStorage.setItem('refreshToken', refreshToken);
      localStorage.setItem('user', JSON.stringify(user));
    },
    logout: (state) => {
      state.accessToken = null;
      state.refreshToken = null;
      state.user = null;
      state.isAuthenticated = false;
      localStorage.removeItem('accessToken');
      localStorage.removeItem('refreshToken');
      localStorage.removeItem('user');
    },
    updateTokens: (state, action) => {
      const { accessToken, refreshToken } = action.payload;
      state.accessToken = accessToken;
      state.refreshToken = refreshToken;
      localStorage.setItem('accessToken', accessToken);
      localStorage.setItem('refreshToken', refreshToken);
    },
    updateUser: (state, action) => {
      state.user = { ...state.user, ...action.payload };
      localStorage.setItem('user', JSON.stringify(state.user));
    },
  },
});
```

### 5.3 Response Interceptor (토큰 자동 갱신)

```javascript
api.interceptors.response.use(
  (response) => response,
  async (error) => {
    const originalRequest = error.config;

    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;

      const refreshToken = localStorage.getItem('refreshToken');
      if (refreshToken) {
        const response = await axios.post('/auth/refresh', { refresh_token: refreshToken });
        const { accessToken, refreshToken: newRefreshToken } = response.data.data;

        store.dispatch(updateTokens({ accessToken, refreshToken: newRefreshToken }));
        originalRequest.headers.Authorization = `Bearer ${accessToken}`;
        return api(originalRequest);
      }
    }

    if (error.response?.status === 401) {
      store.dispatch(logout());
      window.location.href = '/login';
    }

    return Promise.reject(error);
  }
);
```
